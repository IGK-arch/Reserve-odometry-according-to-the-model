cmake_minimum_required(VERSION 3.16)
project(tram_odometry_portable LANGUAGES CXX)
enable_testing()
set(TRAM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/ros2_ws/src/tram_odometry")
add_library(tram_navigation "${TRAM_ROOT}/src/estimator.cpp" "${TRAM_ROOT}/src/navigation.cpp")
target_compile_features(tram_navigation PUBLIC cxx_std_17)
target_include_directories(tram_navigation PUBLIC "${TRAM_ROOT}/include")
target_compile_options(tram_navigation PRIVATE -Wall -Wextra -Wpedantic)
add_executable(navigation_replay evaluation/navigation_replay.cpp)
target_link_libraries(navigation_replay PRIVATE tram_navigation)
add_executable(replay_cli evaluation/replay_cli.cpp)
target_link_libraries(replay_cli PRIVATE tram_navigation)
file(GLOB TRAM_TESTS CONFIGURE_DEPENDS "${TRAM_ROOT}/test/*_test.cpp")
foreach(source IN LISTS TRAM_TESTS)
  get_filename_component(name "${source}" NAME_WE)
  add_executable("${name}" "${source}")
  target_link_libraries("${name}" PRIVATE tram_navigation)
  # Existing standalone tests use assert, including in release builds.
  target_compile_options("${name}" PRIVATE -UNDEBUG)
  if(name STREQUAL "route_tf_test" OR name STREQUAL "stop_landmark_test")
    add_test(NAME "${name}" COMMAND "${name}" "${TRAM_ROOT}/assets/route_map.csv")
  elseif(name STREQUAL "elevation_profile_test")
    add_test(NAME "${name}" COMMAND "${name}" "${TRAM_ROOT}/assets/official_elevation.csv")
  else()
    add_test(NAME "${name}" COMMAND "${name}")
  endif()
endforeach()
